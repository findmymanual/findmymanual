function exportAllFilesLinksToSheet() {
  var folderId = "1kLC6WMOfJnxEr5l8cy7C_waZbp5eyiRi"; // Replace with your folder ID
  var folder = DriveApp.getFolderById(folderId);
  var files = folder.getFiles();
  
  // Create a new Google Sheet
  var ss = SpreadsheetApp.create("Drive File Links");
  var sheet = ss.getActiveSheet();
  sheet.appendRow(["File Name", "URL"]); // Header row

  while (files.hasNext()) {
    var file = files.next();
    
    // Make file public if not already
    if (file.getSharingAccess() !== DriveApp.Access.ANYONE_WITH_LINK) {
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    }

    // Append name and URL to sheet
    sheet.appendRow([file.getName(), file.getUrl()]);
  }

  Logger.log("All links exported! Sheet URL: " + ss.getUrl());
}
